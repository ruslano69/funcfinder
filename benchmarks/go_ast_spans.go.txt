// Native syntax reference for the experimental D Go finder, no build-tag filtering.
package main
import("encoding/json";"go/ast";"go/parser";"go/token";"os")
type symbol struct{Name string `json:"name"`;Start int `json:"start"`;NameLine int `json:"name_line"`;End int `json:"end"`;Bodyless bool `json:"bodyless"`}
type file struct{Path string `json:"path"`;Errors []string `json:"errors"`;Functions []symbol `json:"functions"`;Classes []symbol `json:"classes"`}
func main(){b,err:=os.ReadFile(os.Args[1]);if err!=nil{panic(err)};var paths []string;if err=json.Unmarshal(b,&paths);err!=nil{panic(err)};files:=make([]file,0,len(paths));for _,path:=range paths{r:=file{Path:path,Functions:[]symbol{},Classes:[]symbol{}};fs:=token.NewFileSet();f,e:=parser.ParseFile(fs,path,nil,parser.SkipObjectResolution|parser.AllErrors);if e!=nil{r.Errors=[]string{e.Error()};files=append(files,r);continue};ast.Inspect(f,func(n ast.Node)bool{switch d:=n.(type){case *ast.FuncDecl:r.Functions=append(r.Functions,symbol{d.Name.Name,fs.Position(d.Pos()).Line,fs.Position(d.Name.Pos()).Line,fs.Position(d.End()).Line,d.Body==nil});case *ast.TypeSpec:r.Classes=append(r.Classes,symbol{d.Name.Name,fs.Position(d.Pos()).Line,fs.Position(d.Name.Pos()).Line,fs.Position(d.End()).Line,false})};return true});files=append(files,r)};if err=json.NewEncoder(os.Stdout).Encode(map[string]any{"files":files});err!=nil{panic(err)}}
