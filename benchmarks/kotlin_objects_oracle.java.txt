import com.intellij.openapi.Disposable;
import com.intellij.openapi.util.Disposer;
import com.intellij.psi.PsiErrorElement;
import com.intellij.psi.util.PsiTreeUtil;
import org.jetbrains.kotlin.cli.jvm.compiler.EnvironmentConfigFiles;
import org.jetbrains.kotlin.cli.jvm.compiler.KotlinCoreEnvironment;
import org.jetbrains.kotlin.config.CompilerConfiguration;
import org.jetbrains.kotlin.psi.*;
import org.jetbrains.kotlin.lexer.KotlinLexer;
import org.jetbrains.kotlin.lexer.KtTokens;
import com.intellij.psi.TokenType;
import java.nio.file.*;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.*;

// Syntax-only reference for JSON editing. Only file/type declaration lists
// are visited. Properties additionally expose their explicit accessors;
// initializer/function/accessor bodies and enum entries remain opaque.
public final class KotlinObjectsOracle {
    static String json(String text) {
        StringBuilder b=new StringBuilder("\"");
        for (int i=0;i<text.length();i++) {
            char c=text.charAt(i);
            if(c=='"'||c=='\\') b.append('\\').append(c);
            else if(c<32)b.append(String.format("\\u%04x",(int)c));
            else b.append(c);
        }
        return b.append('"').toString();
    }
    static void walk(List<KtDeclaration> declarations, String source, List<String> objects) {
        for(KtDeclaration declaration:declarations) {
            String kind=null,name=null;
            if(declaration instanceof KtNamedFunction function) {
                if(function.getNameIdentifier()!=null){kind="function";name=function.getNameIdentifier().getText();}
            } else if(declaration instanceof KtProperty property) {
                if(property.getNameIdentifier()!=null){kind="property";name=property.getNameIdentifier().getText();}
            } else if(declaration instanceof KtPropertyAccessor accessor) {
                kind="accessor";name=accessor.isGetter()?"get":"set";
            } else if(declaration instanceof KtTypeAlias alias) {
                kind="type";name=alias.getNameIdentifier().getText();
            } else if(declaration instanceof KtClassOrObject type && !(type instanceof KtEnumEntry)) {
                if(type.getNameIdentifier()!=null){kind="type";name=type.getNameIdentifier().getText();}
                // An unnamed companion is a scope, not an editable type object.
            }
            if(kind!=null) {
                int start=declaration.getTextRange().getStartOffset(),end=declaration.getTextRange().getEndOffset();
                // PSI can attach leading ordinary comments and trailing whitespace.
                // The edit protocol owns code tokens, leaving edge trivia outside.
                KotlinLexer lexer=new KotlinLexer();lexer.start(source.substring(start,end));
                int first=-1,last=0;
                while(lexer.getTokenType()!=null) {
                    if(lexer.getTokenType()!=TokenType.WHITE_SPACE && !KtTokens.COMMENTS.contains(lexer.getTokenType())) {
                        if(first<0)first=lexer.getTokenStart();last=lexer.getTokenEnd();
                    }
                    lexer.advance();
                }
                end=start+last;start+=first;
                int line=1,lineStart=0;
                for(int i=0;i<start;i++)if(source.charAt(i)=='\n'){line++;lineStart=i+1;}
                int column=source.substring(lineStart,start).getBytes(StandardCharsets.UTF_8).length+1;
                objects.add("{\"kind\":"+json(kind)+",\"name\":"+json(name)+",\"line\":"+line+
                    ",\"column\":"+column+",\"source\":"+json(source.substring(start,end))+"}");
            }
            if(declaration instanceof KtClassOrObject type && !(type instanceof KtEnumEntry))walk(type.getDeclarations(),source,objects);
            if(declaration instanceof KtProperty property)walk(new ArrayList<KtDeclaration>(property.getAccessors()),source,objects);
            if(declaration instanceof KtScript script)walk(script.getDeclarations(),source,objects);
        }
    }
    public static void main(String[] args) throws Exception {
        System.setOut(new java.io.PrintStream(System.out,true,StandardCharsets.UTF_8));
        Disposable disposable=Disposer.newDisposable();
        try {
            KotlinCoreEnvironment environment=KotlinCoreEnvironment.Companion.createForProduction(
                disposable,new CompilerConfiguration(),EnvironmentConfigFiles.JVM_CONFIG_FILES);
            KtPsiFactory factory=new KtPsiFactory(environment.getProject(),false);
            for(String path:Files.readAllLines(Path.of(args[0]),StandardCharsets.UTF_8)) {
                if(path.isEmpty())continue;
                byte[] raw=Files.readAllBytes(Path.of(path));
                String source=new String(raw,StandardCharsets.UTF_8).replace("\r\n","\n");
                KtFile file=factory.createFile(Path.of(path).getFileName().toString(),source);
                List<String> objects=new ArrayList<>(),errors=new ArrayList<>();
                for(PsiErrorElement error:PsiTreeUtil.collectElementsOfType(file,PsiErrorElement.class))errors.add(json(error.getErrorDescription()));
                if(errors.isEmpty())walk(file.getDeclarations(),source,objects);
                System.out.println("{\"path\":"+json(path)+",\"language\":\"kotlin\",\"sha256\":"+
                    json(HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(raw)))+
                    ",\"objects\":["+String.join(",",objects)+"],\"errors\":["+String.join(",",errors)+"]}");
            }
        } finally {Disposer.dispose(disposable);}
    }
}
